\chapter{L-tromino Tiling of Rectangles and Deficient Rectangles}

\section{Basic Definitions}

\begin{definition}
  \label{def:cell}
  \lean{Cell}
  \leanok
  A \emph{cell} is a point $(x,y) \in \mathbb{Z}^2$.
\end{definition}

\begin{definition}
  \label{def:region}
  \lean{Region}
  \leanok
  A \emph{region} is a finite set of cells.
\end{definition}

\begin{definition}
  \label{def:ltromino}
  \lean{lTromino}
  \uses{def:cell}
  \leanok
  An \emph{L-tromino} is the prototile with cells $\{(0,0),(1,0),(0,1)\}$.
  Together with its three rotations it gives four orientations.
\end{definition}

\begin{definition}
  \label{def:ltrominoset}
  \lean{lTrominoSet}
  \uses{def:ltromino}
  \leanok
  The \emph{L-tromino protoset} consists of the L-tromino and all its rotations.
\end{definition}

\begin{definition}
  \label{def:ltileable}
  \lean{LTileable}
  \uses{def:region, def:ltrominoset}
  \leanok
  A region $R$ is \emph{L-tromino tileable} if it can be partitioned into
  translated copies of tiles from the L-tromino protoset.
\end{definition}

\begin{definition}
  \label{def:rectangle}
  \lean{rectangle}
  \uses{def:region}
  \leanok
  The \emph{rectangle} $\mathit{rect}(m,n) = \{(x,y) \mid 0 \le x < m,\; 0 \le y < n\}$.
\end{definition}

\section{Necessary Conditions}

\begin{lemma}
  \label{lem:area_div_3}
  \lean{LTileable.area_div_3}
  \uses{def:ltileable}
  \leanok
  \proven
  If a region is L-tromino tileable then its cardinality is divisible by $3$.
\end{lemma}

\begin{lemma}
  \label{lem:not_1_by_n}
  \lean{not_tileable_1_by_n}
  \uses{def:ltileable, def:rectangle}
  \leanok
  \proven
  No $1 \times n$ rectangle ($n \ge 1$) is L-tromino tileable.
\end{lemma}

\begin{lemma}
  \label{lem:not_n_by_1}
  \lean{not_tileable_n_by_1}
  \uses{def:ltileable, def:rectangle}
  \leanok
  \proven
  No $n \times 1$ rectangle ($n \ge 1$) is L-tromino tileable.
\end{lemma}

\begin{lemma}
  \label{lem:not_3_by_odd}
  \lean{not_tileable_3_by_odd}
  \uses{lem:not_1_by_n, def:ltileable, def:rectangle}
  \leanok
  \proven
  No $3 \times (2k+1)$ rectangle is L-tromino tileable.
\end{lemma}

\begin{lemma}
  \label{lem:not_odd_by_3}
  \lean{not_tileable_odd_by_3}
  \uses{lem:not_n_by_1, def:ltileable, def:rectangle}
  \leanok

\end{lemma}

  \lean{tileable_odd_x_mult3}

  \leanok
  \proven
  For $n \ge 3$ odd and $k \ge 2$, the $n \times 3k$ rectangle is L-tromino tileable.
\end{lemma}

\begin{definition}
  \label{def:recttileable_cond}
  \lean{RectTileableConditions}
  \uses{def:rectangle}
  \leanok
  A rectangle $\mathit{rect}(m,n)$ satisfies the \emph{tileability conditions} if
  $3 \mid mn$, $m,n \ge 2$, and neither $(m = 3$ and $n$ odd$)$ nor $(n = 3$ and $m$ odd$)$.
\end{definition}

\begin{theorem}
  \label{thm:rect_iff}
  \lean{rect_tileable_iff}
  \uses{lem:area_div_3, lem:not_1_by_n, lem:not_3_by_odd,
        def:recttileable_cond, def:rectangle, def:ltileable,
        lem:tileable_odd_x_mult3, lem:tileable_even_mult3}
  \leanok
  \proven
  A rectangle $\mathit{rect}(m,n)$ is L-tromino tileable if and only if
  it satisfies the tileability conditions.
\end{theorem}

\section{Rectangle Minus One Corner}

\begin{definition}
  \label{def:rect_minus_corner}
  \lean{rectangleMinusCorner}
  \uses{def:rectangle}
  \leanok
  $R^-(m,n) = \mathit{rect}(m,n) \setminus \{(m-1,n-1)\}$,
  the rectangle with its top-right corner removed.
\end{definition}

\begin{lemma}
  \label{lem:rmc_area}
  \lean{rectMinusCorner_tileable_area_div_3}
  \uses{def:rect_minus_corner, def:ltileable}
  \leanok
  \proven
  If $R^-(m,n)$ is L-tromino tileable then $3 \mid (mn-1)$,
  i.e.\ $mn \equiv 1 \pmod{3}$.
\end{lemma}

\begin{lemma}

  \leanok

\begin{lemma}
  \label{lem:tileable_rmc_mod2_jk_ge2}
  \lean{tileable_rectMinusCorner_mod2_jk_ge2}

  \proven
  For $j,k \ge 2$, $R^-(3j+2,\,3k+2)$ is L-tromino tileable.
  Split vertically: bottom $(3j+2) \times 3k$ rectangle plus top $R^-(3j+2,\,2)$.
\end{lemma}

\begin{lemma}
  \label{lem:tileable_rmc_mod1_jk_ge}
  \lean{tileable_rectMinusCorner_mod1_jk_ge}
  \uses{lem:tileable_4x_rmc_3kplus1, thm:rect_iff, def:rect_minus_corner}
  \leanok
  \proven
  For $j \ge 3$, $k \ge 1$, $R^-(3k+1,\,3j+1)$ is L-tromino tileable.
  Split vertically: bottom $(3k+1) \times 3(j-1)$ rectangle plus top $R^-(3k+1,\,4)$.
\end{lemma}

\begin{lemma}
  \label{lem:tileable_rmc_mod2_case}
  \lean{tileable_rectMinusCorner_mod2_case}
  \uses{lem:tileable_rmc_mod2_jk_ge2, lem:tileable_3kplus2_x2_minus,
        lem:tileable_5x_3kplus2_minus, lem:tileable_5x5_minus,
        lem:tileable_2x2_minus, def:rect_minus_corner}
  \leanok
  \proven
  For $m,n \ge 2$ with $m \equiv n \equiv 2 \pmod{3}$ and $m \le n$,
  $R^-(m,n)$ is L-tromino tileable. (Full case split on $j,k$.)
\end{lemma}

\begin{lemma}
  \label{lem:tileable_rmc_mod1_case}
  \lean{tileable_rectMinusCorner_mod1_case}
  \uses{lem:tileable_rmc_mod1_jk_ge, lem:tileable_4x_rmc_3kplus1,
        lem:tileable_7x7_minus, lem:tileable_4x4_minus,
        lem:tileable_4x7_minus, def:rect_minus_corner}
  \leanok
  \proven
  For $m,n \ge 2$ with $m \equiv n \equiv 1 \pmod{3}$ and $m \le n$,
  $R^-(m,n)$ is L-tromino tileable. (Full case split on $k, j$.)
\end{lemma}

\begin{theorem}
  \label{thm:rmc_iff}
  \lean{rectMinusCorner_tileable_iff}
  \uses{lem:rmc_area, lem:tileable_rmc_mod2_case, lem:tileable_rmc_mod1_case,
        def:rect_minus_corner, def:ltileable, thm:rect_iff}
  \leanok
  \proven
  For $m,n \ge 2$: $R^-(m,n)$ is L-tromino tileable if and only if $mn \equiv 1 \pmod{3}$.
\end{theorem}

\section{Rectangle Minus Two Opposite Corners}

\begin{definition}
  \label{def:rect_minus2}
  \lean{rectangleMinus2Corner}
  \uses{def:rectangle}
  \leanok
  $R^{--}(m,n) = \mathit{rect}(m,n) \setminus \{(0,0),\,(m-1,n-1)\}$,
  the rectangle with both $(0,0)$ and the top-right corner removed.
\end{definition}

\begin{lemma}
  \label{lem:mod3_cases}
  \lean{mod3_one_two_or_two_one_of_area_mod3_two}
  \leanok
  \proven
  If $mn \equiv 2 \pmod{3}$ then either $m \equiv 1,\; n \equiv 2 \pmod{3}$
  or $m \equiv 2,\; n \equiv 1 \pmod{3}$.
\end{lemma}

\begin{lemma}
  \label{lem:single_ltromino}
  \lean{LTileable_single_lTromino}
  \uses{def:ltileable}
  \leanok
  \proven
  Any single placed L-tromino defines an L-tromino tileable region.
\end{lemma}

\begin{lemma}
  \label{lem:topright_ltromino}
  \lean{LTileable_topRightLTromino}
  \uses{lem:single_ltromino, def:ltileable}
  \leanok
  \proven
  For any $j,k$, the top-right L-tromino region in $R^{--}(3j+1, 3k+2)$ is tileable.
\end{lemma}

\begin{lemma}
  \label{lem:3jplus2_3kplus1}
  \lean{tileable_rectangleMinus2Corner_3jplus2_3kplus1}
  \uses{def:rect_minus2, def:ltileable, thm:rmc_iff, lem:mod3_cases,
        thm:rect_iff, lem:topright_ltromino}
  \leanok
  \proven
  $R^{--}(3j+2,\; 3k+1)$ is L-tromino tileable for all $j, k \ge 0$.
\end{lemma}

\begin{lemma}
  \label{lem:4_3kplus2}
  \lean{tileable_rectangleMinus2Corner_4_3kplus2}
  \uses{def:rect_minus2, def:ltileable, thm:rmc_iff}
  \leanok
  \proven
  $R^{--}(4,\; 3k+2)$ is L-tromino tileable for all $k \ge 0$.
\end{lemma}

\begin{lemma}
  \label{lem:3jplus1_3kplus2}
  \lean{tileable_rectangleMinus2Corner_3jplus1_3kplus2}
  \uses{lem:4_3kplus2, lem:3jplus2_3kplus1, def:rect_minus2, def:ltileable,
        thm:rect_iff}
  \leanok
  \proven
  $R^{--}(3j+1,\; 3k+2)$ is L-tromino tileable for all $j, k \ge 0$.
\end{lemma}

\begin{theorem}
  \label{thm:rmc2_iff}
  \lean{rectMinus2Corner_tileable_of_area_mod2}
  \uses{lem:3jplus2_3kplus1, lem:3jplus1_3kplus2, lem:mod3_cases,
        def:rect_minus2, def:ltileable}
  \leanok
  \proven
  $R^{--}(m,n)$ is L-tromino tileable whenever $mn \equiv 2 \pmod{3}$
  (and $m,n \ge 3$).
\end{theorem}
